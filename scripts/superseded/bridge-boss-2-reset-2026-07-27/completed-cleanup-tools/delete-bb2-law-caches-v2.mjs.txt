const digest = (bytes) => createHash("sha256").update(bytes).digest("hex");

async function exists(target) {
  try {
    await stat(target);
    return true;
  } catch (error) {
    if (error.code === "ENOENT") return false;
    throw error;
  }
}

async function exactFile(relative, sha256) {
  const bytes = await readFile(absolute(relative));
  const actual = digest(bytes);
  if (actual !== sha256) throw new Error(`Preserved-file mismatch: ${relative} -> ${actual}`);
  return bytes;
}

async function realDirectory(target, expectedParent = null) {
  const info = await lstat(target);
  if (!info.isDirectory() || info.isSymbolicLink()) {
    throw new Error(`Not a real directory: ${target}`);
  }
  const resolved = await realpath(target);
  if (expectedParent !== null && !winPathEqual(path.dirname(resolved), expectedParent)) {
    throw new Error(`Real-path containment failure: ${target} -> ${resolved}`);
  }
  return resolved;
}

async function members(root, current = root) {
  const entries = await readdir(current, { withFileTypes: true });
  const result = [];
  for (const entry of entries) {
    const target = path.join(current, entry.name);
    const info = await lstat(target);
    if (info.isSymbolicLink()) throw new Error(`Linked cache member forbidden: ${target}`);
    if (entry.isDirectory()) result.push(...(await members(root, target)));
    else if (entry.isFile()) {
      const bytes = await readFile(target);
      result.push({
        path: path.relative(root, target).split(path.sep).join("/"),
        bytes: bytes.length,
        sha256: digest(bytes),
      });
    } else throw new Error(`Unsupported cache member: ${target}`);
  }
  return result.sort((a, b) => (a.path < b.path ? -1 : a.path > b.path ? 1 : 0));
}

function assertTarget(target) {
  if (!winPathEqual(path.dirname(target), parent) || !names.includes(path.basename(target))) {
    throw new Error(`Target is outside the exact allowlist: ${target}`);
  }
}

async function verifyTarget(target, expected, parentReal, allowExactSubset = false) {
  assertTarget(target);
  await realDirectory(target, parentReal);
  const actual = await members(target);
  if (!allowExactSubset) {
    if (JSON.stringify(actual) !== JSON.stringify(expected.records)) {
      throw new Error(`Cache membership mismatch: ${target}`);
    }
    return { files: actual.length, partial: false };
  }
  const expectedByPath = new Map(expected.records.map((record) => [record.path, record]));
  for (const record of actual) {
    const expectedRecord = expectedByPath.get(record.path);
    if (!expectedRecord || JSON.stringify(record) !== JSON.stringify(expectedRecord)) {
      throw new Error(`Partial cache contains a new or changed member: ${target}/${record.path}`);
    }
  }
  return { files: actual.length, partial: actual.length !== expected.records.length };
}

async function states(targets) {
  return Promise.all(
    targets.map(async ({ name, target }) => ({ name, present: await exists(target) })),
  );
}

async function verifyRetirements() {
  for (const item of retirements) {
    const sourcePresent = await exists(absolute(item.source));
    const destinationPresent = await exists(absolute(item.destination));
    if (sourcePresent === destinationPresent) {
      throw new Error(`Retirement must exist at exactly one path: ${item.source}`);
    }
    const file = sourcePresent ? item.source : item.destination;
    await exactFile(file, item.sha256);
  }
}

function validateCompletion(value) {
  const keys = [
    "completed",
    "deletedFiles",
    "deletedTrees",
    "payloadRollbackAvailable",
    "planSha256",
    "publicationOrDeploymentPerformed",
    "reclaimedLogicalBytes",
    "retainedManifestSha256",
    "retirements",
    "schemaVersion",
    "status",
  ].sort();
  if (
    JSON.stringify(Object.keys(value).sort()) !== JSON.stringify(keys) ||
    value.schemaVersion !== 2 ||
    value.status !== "complete" ||
    Number.isNaN(Date.parse(value.completed)) ||
    value.planSha256 !== planSha256 ||
    value.deletedTrees !== 5 ||
    value.deletedFiles !== 1400 ||
    value.reclaimedLogicalBytes !== 2398775787 ||
    value.payloadRollbackAvailable !== false ||
    value.retainedManifestSha256 !== expectedManifest ||
    JSON.stringify(value.retirements) !== JSON.stringify(retirements) ||
    value.publicationOrDeploymentPerformed !== false
  ) {
    throw new Error("Completion contract mismatch.");
  }
}
