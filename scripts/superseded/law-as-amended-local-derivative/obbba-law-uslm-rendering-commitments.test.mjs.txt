import assert from "node:assert/strict";
import { mkdtemp, readFile, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import test from "node:test";

import {
  buildObbbaLawUslmRenderingCommitments,
  commitRenderingCommitmentsCreateOnly,
  runObbbaLawUslmRenderingCommitmentsWithCommitments,
  validateRenderingCommitments,
} from "../build-obbba-law-uslm-rendering-commitments.mjs";

const HASH = "a".repeat(64);
const expected = {
  expectedTargets: 1,
  expectedRenderings: 1,
  pagePlanSha256: HASH,
  storeEvidenceSha256: HASH,
};

function fixture() {
  return {
    version: 1,
    mode: "obbba-law-uslm-rendering-commitments",
    authorityBoundary:
      "This artifact commits to reviewed rendering metadata only. It contains "
      + "no legal text, XML, Markdown body, or editorial prose.",
    inputs: {
      pagePlanSha256: HASH,
      sectionStoreEvidenceSha256: HASH,
    },
    summary: {
      targets: 1,
      renderings: 1,
      roles: { before: 0, current: 1 },
    },
    entries: [{
      uscTitle: "7",
      uscSection: "1",
      role: "current",
      sourceFragmentSha256: HASH,
      canonicalAstSha256: HASH,
      operativeTextCommitment: {
        profile: "uslm-operative-text-v1",
        characters: 1,
        words: 1,
        sha256: HASH,
      },
      visibleTextCommitment: {
        profile: "obbba-uslm-visible-text-v1",
        characters: 1,
        words: 1,
        sha256: HASH,
      },
      bodySha256: HASH,
      referenceMapProfile: "obbba-official-reference-map-v1",
      references: [],
      tables: [],
    }],
  };
}

test("validates strict text-free rendering commitments", () => {
  validateRenderingCommitments(fixture(), expected);
  for (const mutate of [
    (value) => { value.extra = true; },
    (value) => { value.entries[0].body = "canary legal text"; },
    (value) => { value.entries[0].visibleTextCommitment.sha256 = "bad"; },
    (value) => { value.entries[0].role = "before"; },
    (value) => { value.summary.targets = 2; },
    (value) => { value.entries.push(structuredClone(value.entries[0])); },
  ]) {
    const value = fixture();
    mutate(value);
    assert.throws(() => validateRenderingCommitments(value, expected));
  }
});

test("rejects recomputed reference and table metadata drift", () => {
  const reference = fixture();
  reference.entries[0].references.push({
    profile: "obbba-official-reference-map-v1",
    originalHref: "/us/usc/t7/s2",
    namespace: "usc",
    releaseRole: "current",
    resolution: "verified-official-link",
    mappedUrl: "https://attacker.example/",
    authority: "Office of the Law Revision Counsel",
  });
  assert.throws(
    () => validateRenderingCommitments(reference, expected),
    /reference commitment/,
  );
  const table = fixture();
  table.entries[0].tables.push({
    columns: [],
    rows: [[{ kind: "td", colspan: "99" }]],
  });
  assert.throws(
    () => validateRenderingCommitments(table, expected),
    /table cell/,
  );
});

test("existing output stops the runner before any input read", async () => {
  const root = await mkdtemp(join(tmpdir(), "obbba-rendering-"));
  try {
    const outputPath = join(root, "competitor.json");
    await writeFile(outputPath, "competitor", "utf8");
    await assert.rejects(
      runObbbaLawUslmRenderingCommitmentsWithCommitments({
        outputPath,
        pagePlanPath: join(root, "missing-plan.json"),
        storeEvidencePath: join(root, "missing-store.json"),
        sectionStoreDirectory: join(root, "missing-sections"),
        expectedStoreBasename: "missing-sections",
        pagePlanSha256: HASH,
        storeEvidenceSha256: HASH,
        expectedTargets: 1,
        expectedRenderings: 1,
      }),
      /output already exists/,
    );
    assert.equal(await readFile(outputPath, "utf8"), "competitor");
  } finally {
    await rm(root, { recursive: true, force: true });
  }
});

test("production wrapper is zero-argument authority with no commit hook", () => {
  assert.equal(buildObbbaLawUslmRenderingCommitments.length, 0);
  assert.doesNotMatch(
    buildObbbaLawUslmRenderingCommitments.toString(),
    /beforeCommit|\.\.\.options/,
  );
});

test("post-validation mutation cannot reach the output", async () => {
  const root = await mkdtemp(join(tmpdir(), "obbba-rendering-mutation-"));
  try {
    const outputPath = join(root, "output.json");
    await assert.rejects(
      commitRenderingCommitmentsCreateOnly(
        fixture(),
        outputPath,
        (value) => {
          value.entries[0].bodySha256 = "b".repeat(64);
          value.entries[0].references.length = 0;
        },
      ),
      /changed after final validation/,
    );
    await assert.rejects(readFile(outputPath), { code: "ENOENT" });
  } finally {
    await rm(root, { recursive: true, force: true });
  }
});

test("late wx collision preserves the competing output", async () => {
  const root = await mkdtemp(join(tmpdir(), "obbba-rendering-race-"));
  try {
    const outputPath = join(root, "output.json");
    await assert.rejects(
      commitRenderingCommitmentsCreateOnly(
        fixture(),
        outputPath,
        async () => writeFile(outputPath, "competitor", "utf8"),
      ),
      { code: "EEXIST" },
    );
    assert.equal(await readFile(outputPath, "utf8"), "competitor");
  } finally {
    await rm(root, { recursive: true, force: true });
  }
});
