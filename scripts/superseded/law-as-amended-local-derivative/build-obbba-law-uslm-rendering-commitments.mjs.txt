import { createHash } from "node:crypto";
import { readFile, writeFile } from "node:fs/promises";
import { basename, isAbsolute, relative, resolve } from "node:path";
import { fileURLToPath, pathToFileURL } from "node:url";

import {
  validateUscSectionStoreEvidence,
} from "./build-obbba-law-usc-comparison-index.mjs";
import {
  buildUslmOperativeAst,
} from "./obbba-law-uslm-ast-lib.mjs";
import {
  renderUslmAstMarkdown,
} from "./obbba-law-uslm-markdown-lib.mjs";
import {
  mapOfficialReference,
} from "./obbba-law-official-reference-map.mjs";

const ROOT = fileURLToPath(new URL("../", import.meta.url));
const PAGE_PLAN_SHA256 =
  "b2b5abde4e24d878501c007faf0e460d3a7badcabb0272c4728b29580a1e243b";
const STORE_SHA256 =
  "90e1f677227f510fd2b25f370d200ba49d5cc6476706d3be50bad66092177cdd";
const STORE_BASENAME = "obbba-law-usc-selected-sections-2026-07-26";
const ROLES = ["before", "current"];

export async function buildObbbaLawUslmRenderingCommitments({
  onProgress,
} = {}) {
  return runObbbaLawUslmRenderingCommitmentsWithCommitments({
    onProgress,
    pagePlanPath: resolve(
      ROOT,
      "docs/evidence/OBBBA_LAW_AS_AMENDED_PAGE_INPUT_PLAN_2026-07-26.json",
    ),
    pagePlanSha256: PAGE_PLAN_SHA256,
    storeEvidencePath: resolve(
      ROOT,
      "docs/evidence/OBBBA_LAW_USC_VERSIONED_SECTION_STORE_2026-07-26.json",
    ),
    storeEvidenceSha256: STORE_SHA256,
    sectionStoreDirectory: resolve(
      ROOT,
      ".discussionbridge-cache",
      STORE_BASENAME,
    ),
    expectedStoreBasename: STORE_BASENAME,
    expectedTargets: 313,
    expectedRenderings: 590,
    outputPath: resolve(
      ROOT,
      "docs/evidence/OBBBA_LAW_USLM_RENDERING_COMMITMENTS_2026-07-26.json",
    ),
  });
}

export async function runObbbaLawUslmRenderingCommitmentsWithCommitments(
  options,
) {
  const outputPath = resolve(options.outputPath);
  await assertAbsent(outputPath);
  if (basename(resolve(options.sectionStoreDirectory))
      !== options.expectedStoreBasename) {
    throw new Error("USLM rendering store basename does not match review.");
  }
  const [pagePlanBytes, storeBytes] = await Promise.all([
    readPinned(options.pagePlanPath, options.pagePlanSha256, "page plan"),
    readPinned(
      options.storeEvidencePath,
      options.storeEvidenceSha256,
      "section store",
    ),
  ]);
  const pagePlan = JSON.parse(pagePlanBytes.toString("utf8"));
  const store = JSON.parse(storeBytes.toString("utf8"));
  validateUscSectionStoreEvidence(store);
  const eligible = eligibleTargetSet(pagePlan);
  if (eligible.size !== options.expectedTargets) {
    throw new Error("USLM rendering eligible target count drifted.");
  }
  const storeByIdentity = new Map(store.entries.map((entry) => [
    `${entry.uscTitle}:${entry.uscSection}`,
    entry,
  ]));
  if ([...eligible].some((identity) => !storeByIdentity.has(identity))) {
    throw new Error("USLM rendering target is absent from the reviewed store.");
  }
  const storeRoot = resolve(options.sectionStoreDirectory, "sections");
  const entries = [];
  for (const identity of [...eligible].sort(identitySort)) {
    const stored = storeByIdentity.get(identity);
    for (const role of ROLES) {
      const version = stored.versions[role];
      if (version.state === "absent") continue;
      const expectedRelative =
        `${role}/usc${stored.uscTitle.padStart(2, "0")}-s${stored.uscSection}.xml`;
      if (version.relativePath !== expectedRelative) {
        throw new Error(`USLM rendering fragment path drift: ${role}:${identity}.`);
      }
      const fragmentPath = resolve(
        storeRoot,
        ...version.relativePath.split("/"),
      );
      requireContained(storeRoot, fragmentPath);
      const fragmentBytes = await readFile(fragmentPath);
      if (fragmentBytes.length !== version.bytes
        || sha256(fragmentBytes) !== version.sha256) {
        throw new Error(`USLM rendering fragment drift: ${role}:${identity}.`);
      }
      const ast = buildUslmOperativeAst(fragmentBytes.toString("utf8"), {
        expectedTitle: stored.uscTitle,
        expectedSection: stored.uscSection,
      });
      const rendered = renderUslmAstMarkdown(ast, { releaseRole: role });
      entries.push(projectCommitment(stored, role, version, rendered));
      await options.onProgress?.({
        completed: entries.length,
        total: options.expectedRenderings,
      });
    }
  }
  if (entries.length !== options.expectedRenderings) {
    throw new Error("USLM rendering commitment count drifted.");
  }
  const result = {
    version: 1,
    mode: "obbba-law-uslm-rendering-commitments",
    authorityBoundary:
      "This artifact commits to reviewed rendering metadata only. It contains "
      + "no legal text, XML, Markdown body, or editorial prose.",
    inputs: {
      pagePlanSha256: options.pagePlanSha256,
      sectionStoreEvidenceSha256: options.storeEvidenceSha256,
    },
    summary: {
      targets: eligible.size,
      renderings: entries.length,
      roles: Object.fromEntries(ROLES.map((role) => [
        role,
        entries.filter((entry) => entry.role === role).length,
      ])),
    },
    entries,
  };
  validateRenderingCommitments(result, {
    expectedTargets: options.expectedTargets,
    expectedRenderings: options.expectedRenderings,
    pagePlanSha256: options.pagePlanSha256,
    storeEvidenceSha256: options.storeEvidenceSha256,
  });
  await commitRenderingCommitmentsCreateOnly(
    result,
    outputPath,
    options.beforeCommit,
  );
  return result;
}

export async function commitRenderingCommitmentsCreateOnly(
  result,
  outputPath,
  beforeCommit,
) {
  const reviewedBytes = `${JSON.stringify(result, null, 2)}\n`;
  await beforeCommit?.(result);
  const finalBytes = `${JSON.stringify(result, null, 2)}\n`;
  if (finalBytes !== reviewedBytes) {
    throw new Error(
      "USLM rendering commitments changed after final validation.",
    );
  }
  await writeFile(resolve(outputPath), finalBytes, {
    encoding: "utf8",
    flag: "wx",
  });
}

export function validateRenderingCommitments(value, expected) {
  exactKeys(value, [
    "authorityBoundary", "entries", "inputs", "mode", "summary", "version",
  ]);
  exactKeys(value.inputs, [
    "pagePlanSha256", "sectionStoreEvidenceSha256",
  ]);
  exactKeys(value.summary, ["renderings", "roles", "targets"]);
  exactKeys(value.summary.roles, ["before", "current"]);
  if (value.version !== 1
    || value.mode !== "obbba-law-uslm-rendering-commitments"
    || value.authorityBoundary
      !== "This artifact commits to reviewed rendering metadata only. It "
        + "contains no legal text, XML, Markdown body, or editorial prose."
    || value.inputs?.pagePlanSha256 !== expected.pagePlanSha256
    || value.inputs?.sectionStoreEvidenceSha256
      !== expected.storeEvidenceSha256
    || value.summary?.targets !== expected.expectedTargets
    || value.summary?.renderings !== expected.expectedRenderings
    || !Array.isArray(value.entries)
    || value.entries.length !== expected.expectedRenderings) {
    throw new Error("USLM rendering commitments shape/counts are invalid.");
  }
  const identities = new Set();
  const targets = new Set();
  const roles = { before: 0, current: 0 };
  for (const entry of value.entries) {
    exactKeys(entry, [
      "bodySha256", "canonicalAstSha256", "operativeTextCommitment",
      "referenceMapProfile", "references", "role", "sourceFragmentSha256",
      "tables", "uscSection", "uscTitle", "visibleTextCommitment",
    ]);
    const identity = `${entry.uscTitle}:${entry.uscSection}:${entry.role}`;
    if (!/^\d+$/.test(entry.uscTitle)
      || !/^[A-Za-z0-9](?:[A-Za-z0-9.-]*[A-Za-z0-9])?$/.test(
        entry.uscSection,
      )
      || !ROLES.includes(entry.role)
      || identities.has(identity)) {
      throw new Error("USLM rendering commitment identity is invalid.");
    }
    identities.add(identity);
    targets.add(`${entry.uscTitle}:${entry.uscSection}`);
    roles[entry.role] += 1;
    for (const hash of [
      entry.bodySha256,
      entry.canonicalAstSha256,
      entry.sourceFragmentSha256,
      entry.operativeTextCommitment?.sha256,
      entry.visibleTextCommitment?.sha256,
    ]) {
      if (!/^[a-f0-9]{64}$/.test(hash ?? "")) {
        throw new Error("USLM rendering commitment hash is invalid.");
      }
    }
    if (entry.operativeTextCommitment.profile !== "uslm-operative-text-v1"
      || entry.visibleTextCommitment.profile
        !== "obbba-uslm-visible-text-v1"
      || entry.referenceMapProfile !== "obbba-official-reference-map-v1"
      || !Array.isArray(entry.references) || !Array.isArray(entry.tables)) {
      throw new Error("USLM rendering commitment profile is invalid.");
    }
    validateTextCommitment(entry.operativeTextCommitment);
    validateTextCommitment(entry.visibleTextCommitment);
    for (const reference of entry.references) {
      if (JSON.stringify(reference) !== JSON.stringify(mapOfficialReference(
        reference?.originalHref,
        { releaseRole: entry.role },
      ))) {
        throw new Error("USLM rendering reference commitment is invalid.");
      }
    }
    for (const table of entry.tables) validateTableCommitment(table);
  }
  if (targets.size !== expected.expectedTargets
    || roles.before !== value.summary.roles?.before
    || roles.current !== value.summary.roles?.current
    || roles.before + roles.current !== expected.expectedRenderings
    || /"(?:body|raw|xml|legalText|editorialText)"\s*:/i.test(
      JSON.stringify(value),
    )) {
    throw new Error("USLM rendering commitment summary/content is invalid.");
  }
}

function eligibleTargetSet(plan) {
  if (!plan || plan.version !== 1
    || plan.mode !== "obbba-law-as-amended-page-input-plan"
    || plan.summary?.pages !== 309
    || !Array.isArray(plan.entries) || plan.entries.length !== 309) {
    throw new Error("USLM rendering page plan is invalid.");
  }
  const targets = new Set();
  for (const entry of plan.entries) {
    if (entry.outcome === "review-required" || entry.sectionId === "20009") {
      continue;
    }
    for (const target of [
      ...(entry.officialLawTargets ?? []),
      ...(entry.contextTargets ?? []),
    ]) {
      if (!/^\d+$/.test(target.uscTitle ?? "")
        || !/^[A-Za-z0-9](?:[A-Za-z0-9.-]*[A-Za-z0-9])?$/.test(
          target.uscSection ?? "",
        )) {
        throw new Error("USLM rendering page target identity is invalid.");
      }
      targets.add(`${target.uscTitle}:${target.uscSection}`);
    }
  }
  return targets;
}

function projectCommitment(stored, role, version, rendered) {
  return JSON.parse(JSON.stringify({
    uscTitle: stored.uscTitle,
    uscSection: stored.uscSection,
    role,
    sourceFragmentSha256: version.sha256,
    canonicalAstSha256: rendered.canonicalAstSha256,
    operativeTextCommitment: rendered.operativeTextCommitment,
    visibleTextCommitment: rendered.visibleTextCommitment,
    bodySha256: rendered.bodySha256,
    referenceMapProfile: rendered.referenceMapProfile,
    references: rendered.references,
    tables: rendered.tables,
  }));
}

function validateTextCommitment(value) {
  exactKeys(value, ["characters", "profile", "sha256", "words"]);
  if (!Number.isSafeInteger(value.characters) || value.characters < 0
    || !Number.isSafeInteger(value.words) || value.words < 0) {
    throw new Error("USLM rendering text commitment is invalid.");
  }
}

function validateTableCommitment(table) {
  const allowed = ["columns", "rows", "width", "xmlns"];
  if (!table || typeof table !== "object" || Array.isArray(table)
    || Object.keys(table).some((key) => !allowed.includes(key))
    || !Array.isArray(table.columns) || !Array.isArray(table.rows)
    || (table.width !== undefined && table.width !== "50%")
    || (table.xmlns !== undefined
      && table.xmlns !== "http://www.w3.org/1999/xhtml")) {
    throw new Error("USLM rendering table commitment is invalid.");
  }
  for (const column of table.columns) {
    if (!column || typeof column !== "object" || Array.isArray(column)
      || Object.keys(column).some((key) => key !== "width")
      || (column.width !== undefined && column.width !== "50%")) {
      throw new Error("USLM rendering table column is invalid.");
    }
  }
  for (const row of table.rows) {
    if (!Array.isArray(row) || !row.length) {
      throw new Error("USLM rendering table row is invalid.");
    }
    for (const cell of row) {
      const keys = Object.keys(cell ?? {});
      if (!cell || typeof cell !== "object" || Array.isArray(cell)
        || keys.some((key) =>
          !["colspan", "kind", "rowspan"].includes(key))
        || !["th", "td"].includes(cell.kind)
        || (cell.colspan !== undefined
          && !["3", "4"].includes(cell.colspan))
        || (cell.rowspan !== undefined && cell.rowspan !== "2")) {
        throw new Error("USLM rendering table cell is invalid.");
      }
    }
  }
}

function requireContained(root, candidate) {
  const rel = relative(root, candidate);
  if (!rel || isAbsolute(rel) || rel === ".."
    || rel.startsWith("../") || rel.startsWith("..\\")) {
    throw new Error("USLM rendering fragment path escapes its store.");
  }
}

async function readPinned(path, expectedSha256, label) {
  if (!/^[a-f0-9]{64}$/.test(expectedSha256 ?? "")) {
    throw new Error(`USLM rendering ${label} commitment is invalid.`);
  }
  const bytes = await readFile(resolve(path));
  if (sha256(bytes) !== expectedSha256) {
    throw new Error(`USLM rendering ${label} bytes do not match review.`);
  }
  return bytes;
}

async function assertAbsent(path) {
  try {
    await readFile(path);
    throw new Error(`USLM rendering output already exists: ${path}.`);
  } catch (error) {
    if (error?.code !== "ENOENT") throw error;
  }
}

function exactKeys(value, keys) {
  if (!value || typeof value !== "object" || Array.isArray(value)
    || Object.keys(value).sort().join(",") !== [...keys].sort().join(",")) {
    throw new Error("USLM rendering commitment entry schema is invalid.");
  }
}

function identitySort(left, right) {
  const [leftTitle, leftSection] = left.split(":");
  const [rightTitle, rightSection] = right.split(":");
  return Number(leftTitle) - Number(rightTitle)
    || leftSection.localeCompare(rightSection, undefined, { numeric: true });
}

function sha256(value) {
  return createHash("sha256").update(value).digest("hex");
}

async function main() {
  const result = await buildObbbaLawUslmRenderingCommitments({
    onProgress: ({ completed, total }) => {
      if (completed === total || completed % 25 === 0) {
        process.stdout.write(`Rendered ${completed}/${total} commitments.\n`);
      }
    },
  });
  process.stdout.write(
    `Rendering commitments: ${result.summary.renderings}.\n`,
  );
}

if (process.argv[1]
  && import.meta.url === pathToFileURL(resolve(process.argv[1])).href) {
  main().catch((error) => {
    process.stderr.write(`${error.message}\n`);
    process.exitCode = 1;
  });
}
